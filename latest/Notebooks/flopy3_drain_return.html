    model_ws=modelpth,
    exe_name="mfnwt",
    version="mfnwt",
)

    m,
    nlay=1,
    nrow=10,
    ncol=10,
    nper=1,
    perlen=1,
    top=10,
    botm=0,
    steady=True,
)

o = flopy.modflow.ModflowOc(m)
